public interface Manageable {
void start();

void stop();

boolean isActive();
}
 

================================


public abstract class CloudResource implements Manageable {
protected String id;
protected String name;
protected boolean status;
protected double loadPercentage;

public CloudResource(String id, String name) {
    this.id = id;
    this.name = name;
    status = false;
    loadPercentage = 0;
}

public void start() {
    status = true;
}

public void stop() {
    status = false;
}

public boolean isActive() {
    return status;
}

public abstract String getDetails();

public String getId() {
    return id;
}

public void setId(String id) {
    this.id = id;
}

public String getName() {
    return name;
}

public void setName(String name) {
    this.name = name;
}

public boolean getStatus() {
    return status;
}

public void setStatus(boolean status) {
    this.status = status;
}

public double getLoadPercentage() {
    return loadPercentage;
}

public void setLoadPercentage(double loadPercentage) {
    this.loadPercentage = loadPercentage;
}
}

import java.util.function.Consumer; import java.util.function.Predicate;
public class ResourcePolicy<T extends CloudResource> {
private String policyName;
private Predicate<T> condition;
private Consumer<T> action;

public ResourcePolicy(String name,
                      Predicate<T> condition,
                      Consumer<T> action) {

    policyName = name;
    this.condition = condition;
    this.action = action;
}

public void apply(T resource) {
    if (condition.test(resource)) {
        action.accept(resource);
    }
}

public String getPolicyName() {
    return policyName;
}
}

import java.util.*; import java.util.stream.Stream;
public class CloudManager {
private Map<String, List<CloudResource>> projects = new HashMap<>();
private List<ResourcePolicy<?>> policies = new ArrayList<>();

public void addResource(String p, CloudResource r) {
    projects.putIfAbsent(p, new ArrayList<>());
    projects.get(p).add(r);
}

public void addPolicy(ResourcePolicy<?> p) {
    policies.add(p);
}

public CloudResource getResourceById(String id) {

    for (List<CloudResource> list : projects.values())
        for (CloudResource r : list)
            if (r.getId().equals(id))
                return r;

    return null;
}

public void applyAllPolicies() {

    for (List<CloudResource> list : projects.
values())
        for (CloudResource r : list)
            for (ResourcePolicy<?> p : policies)
                apply(p, r);
}

private <T extends CloudResource> void apply(
        ResourcePolicy<T> p, CloudResource r) {

    try {
        p.apply((T) r);
    } catch (ClassCastException e) {
        // Политика относится к другому типу ресурса
    }
}

public Stream<CloudResource> getAnalyticsStream() {
    return projects.values()
            .stream()
            .flatMap(List::stream);
}
}

import java.util.Scanner;
public class Main {
public static void main(String[] args) {

    Scanner sc = new Scanner(System.in);
    CloudManager manager = new CloudManager();

    while (true) {

        System.out.println("\n--- CLOUD MANAGER ---");
        System.out.println("1 - Добавить ресурс");
        System.out.println("2 - Запустить/остановить");
        System.out.println("3 - Базовая политика");
        System.out.println("4 - Политика варианта");
        System.out.println("5 - Запустить политики");
        System.out.println("6 - Аналитика");
        System.out.println("7 - Выход");

        int n = sc.nextInt();
        sc.nextLine();

        switch (n) {

            case 1:
                addResource(sc, manager);
                break;

            case 2:
                changeStatus(sc, manager);
                break;

            case 3:
                addBasicPolicy(manager);
                break;

            case 4:
                System.out.println("Здесь будет политика варианта");
                break;

            case 5:
                manager.applyAllPolicies();
                System.out.println("Политики выполнены");
                break;

            case 6:
                System.out.println("Здесь будет аналитика варианта");
                break;

            case 7:
                return;

            default:
                System.out.println("Ошибка");
        }
    }
}

static void addResource(
        Scanner sc, CloudManager manager) {

    System.out.println("1 - VM");
    System.out.println("2 - Database");
    System.out.println("3 - LoadBalancer");

    int type = sc.nextInt();
    sc.nextLine();

    System.out.print("ID: ");
    String id = sc.nextLine();

    System.out.print("Name: ");
    String name = sc.nextLine();

    System.out.print("Project: ");
    String project = sc.nextLine();

    CloudResource r;

    if (type == 1) {

        System.out.print("RAM: ");
        int ram = sc.nextInt();

        r = new VirtualMachine(id, name, ram);

    } else if (type == 2) {

        System.out.print("Storage: ");
        double storage = sc.nextDouble();

        r = new CloudDatabase(id, name, storage);

    } else {

        System.out.print("Connections: ");
        int connections = sc.nextInt();

        r = new LoadBalancer(id, name, connections);
    }

    manager.addResource(project, r);

    System.out.println("Ресурс добавлен");
}

static void changeStatus(
        Scanner sc, CloudManager manager) {

    System.out.print("ID ресурса: ");
    String id = sc.nextLine();

    CloudResource r = manager.getResourceById(id);

    if (r == null) {
        System.out.println("Ресурс не найден");
        return;
    }

    if (r.isActive()) {
        r.stop();
        System.out.println("Ресурс остановлен");
    } else {
        r.start();
        System.out.println("Ресурс запущен");
    }
}

static void addBasicPolicy(CloudManager manager) {

    ResourcePolicy<CloudResource> p =
            new ResourcePolicy<>(
                    "Сброс нагрузки",

                    r -> !r.isActive()
                            && r.getLoadPercentage() > 0,

                    r -> r.setLoadPercentage(0)
            );

    manager.addPolicy(p);

    System.out.println("Политика добавлена");
}
}
