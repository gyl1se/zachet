package interfaces;

public interface Dispatchable {
    void dispatch();
    void recall();
    boolean isInTransit();
}

package models;

import interfaces.Dispatchable;

public abstract class Transport implements Dispatchable {
    protected String id;
    protected String model;
    protected boolean status;
    protected double batteryLevel;

    @Override
    public void dispatch(){
        status = true;
    }

    @Override
    public void recall(){
        status = false;
    }

    @Override
    public boolean isInTransit(){
        return status;
    }

    public Transport(String id,String model,boolean status,double batteryLevel){
        this.id = id;
        this.model = model;
        this.status = status;
        this.batteryLevel = batteryLevel;

    }

    public String getId(){
        return id;
    }
    public void setId(String id){
        this.id = id;
    }

    public String getModel(){
        return model;
    }
    public void setModel(String model){
        this.model = model;
    }

    public boolean getStatus(){
        return status;
    }
    public void setStatus(boolean status){
        this.status = status;
    }

    public double getBatteryLevel(){
        return batteryLevel;
    }
    public void setBatteryLevel(double batteryLevel){
        this.batteryLevel = batteryLevel;
    }

    public abstract String getDetails();

    @Override
    public String toString(){
        return getDetails();
    }
}

package logic;

import interfaces.Dispatchable;
import models.Transport;
import java.util.function.Consumer;
import java.util.function.Predicate;

public class DispatchRule<T extends Transport> {
    private String ruleName;
    private Predicate<T> condition;
    private Consumer<T> action;

    public DispatchRule(String ruleName,Predicate<T> condition,Consumer<T> action){
        this.ruleName = ruleName;
        this.condition = condition;
        this.action = action;
    }

    public String getRuleName(){
        return ruleName;
    }

    public void apply(T transport){
        if (condition.test(transport)){
            action.accept(transport);
        }

    }
}

public class LogisticsManager {
    private Map<String, List<Transport>> hubs;
    private List<DispatchRule<Transport>> rules;

    public LogisticsManager() {
        this.hubs = new HashMap<>();
        this.rules = new ArrayList<>();
    }

    public void addTransport(String hubName,Transport transport){
        hubs.computeIfAbsent(hubName,k -> new ArrayList<>()).add(transport);
    }

    public void addRule(DispatchRule<Transport> rule){
        rules.add(rule);
    }

    public Transport getTransportById(String id){
        return hubs.values().stream().flatMap(List::stream).
                filter(transport -> transport.getId().equalsIgnoreCase(id)).findFirst().orElse(null);
    }

    public void applyAllRules(){
        for (DispatchRule<Transport> rule : rules){
            hubs.values().stream().flatMap(List::stream).forEach(rule::apply);
        }
    }

    public Stream<Transport> getAnalysticsStream(){
        return hubs.values().stream().flatMap(List::stream);
    }


    public void printAllTransports(){
        if (hubs.isEmpty()){
            System.out.println("Транспортов нет");
            return;
        }

        hubs.forEach(((hub, transports) -> {
            transports.forEach(transport -> System.out.println(" "+transport));
        }));
    }

    public static void main(String[] args){

                case 3 -> Batery(manager);

                    manager.applyAllRules();

    public static void toogleTransport(Scanner scanner, LogisticsManager manager){

        System.out.println("Введите ID: " );

        String id = scanner.nextLine();

        Transport transport = manager.getTransportById(id);

        if (transport != null){
            System.out.println("1.Отравить");
            System.out.println("2.Вернуть");

            int choice = Integer.parseInt(scanner.nextLine());
            switch (choice){
                case 1 -> transport.dispatch();
                case 2 -> transport.recall();
            }
        }



        System.out.println("1.Отравить");
        System.out.println("2.Вернуть");


    }

    public static void Special(LogisticsManager manager){
        DispatchRule<Transport> rule = new DispatchRule<>("Экстренный вызов",
                transport -> transport instanceof DeliveryDrone drone && drone.isInTransit() && drone.getBatteryLevel() < 15,
                transport -> {
            DeliveryDrone drone = (DeliveryDrone) transport;
            drone.recall();
                });
        manager.addRule(rule);
    }




    public static void Batery(LogisticsManager manager){
        DispatchRule<Transport> rule = new DispatchRule<>(" Калибровка батареи",
                transport -> transport.getBatteryLevel() < 0,
                transport -> transport.setBatteryLevel(0)
                );
        manager.addRule(rule);
        System.out.println("Правило добавлено");
    }


    public static void addTransport(Scanner scanner, LogisticsManager manager){
        System.out.println("Название хаба: ");
        String hubName = scanner.nextLine();
        System.out.println("Модель: ");
        String model = scanner.nextLine();
        System.out.println("ID: ");
        String id = scanner.nextLine();

        System.out.println("Тип транспорта: ");
        System.out.println("1. DeliveryDrone: ");
        System.out.println("2. CargoHelicopter: ");
        System.out.println("3. GroundRobot: ");
        System.out.println("Введите цифру: ");

        int type = Integer.parseInt(scanner.nextLine());

        System.out.println("Находится в пути: true/false");

        boolean status = Boolean.parseBoolean(scanner.nextLine());

        System.out.println("Уровень заряда: ");

        double level = Double.parseDouble(scanner.nextLine());

        Transport transport = null;

        switch (type){
            case 1 -> {
                System.out.println("MaxPayLoad: ");
                double maxP = Double.parseDouble(scanner.nextLine());
                transport = new DeliveryDrone(id,model,status,level,maxP);
            }
            case 2 -> {
                System.out.println("flightRange: ");
                double flight = Double.parseDouble(scanner.nextLine());
                transport = new CargoHelicopter(id,model,status,level,flight);

            }
            case 3 -> {
                System.out.println("wheelCondition: ");
                int wheel = Integer.parseInt(scanner.nextLine());
                transport = new GroundRobot(id,model,status,level,wheel);

            }

        }

        if (manager != null){
            System.out.println("Транспорт добавлен");
            manager.addTransport(hubName,transport);
        }
    }

    private static void analytics(LogisticsManager manager) {
        List<Transport> top3LowBattery = manager.getAnalysticsStream()
                .sorted(
                        Comparator.comparingDouble(
                                Transport::getBatteryLevel
                        )
                )
                .limit(3)
                .collect(Collectors.toList());

        if (top3LowBattery.isEmpty()) {
            System.out.println("В сети пока нет транспорта.");
            return;
        }

        System.out.println(
                "\n=== ТОП-3 ТРАНСПОРТА С НАИМЕНЬШИМ ЗАРЯДОМ ==="
        );

        top3LowBattery.forEach(transport ->
                System.out.println(transport.getDetails())
        );
    }




}
